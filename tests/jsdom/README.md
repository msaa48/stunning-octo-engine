# اختبارات jsdom (من غير متصفح، بتشتغل في الشات)
```
cd tests/jsdom && npm i jsdom --no-save
node run1.js   # فهم المساعد المحلي لكل دور (طباعة للمراجعة)
node ui.js     # واجهة المساعد: الأدوار، منع أدوات دور تاني، XSS، AI fallback، أوفلاين
node sa.js     # لوحة مفاتيح الـAI عند المالك
node map.js    # خريطة النظام ومركز التحكم والفحص الشامل
```
بتقرا `index.html` وبتقتطع الأجزاء بالـmarkers، فلو غيّرت عنوان تعليق marker لازم تحدّثه هنا.
